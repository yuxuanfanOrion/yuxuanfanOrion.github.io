\newenvironment{pubitem}{%
  \def\pubtagvalue{}\def\pubtitlevalue{}\def\pubauthorsvalue{}%
  \def\publinksvalue{}\def\pubvenuevalue{}\def\pubnotevalue{}\def\pubsummaryvalue{}%
}{%
  \par\addvspace{2.4pt}\noindent
  \begin{minipage}{\linewidth}
    \refstepcounter{publication}%
    \begingroup
      \fontsize{10pt}{11.5pt}\selectfont\raggedright
      \hyphenpenalty=5000\exhyphenpenalty=5000
      \hangindent=\pubindent\hangafter=1
      \makebox[\pubindent][l]{{\fontsize{8.7pt}{10pt}\selectfont\color{cvaccent}\thepublication.}}%
      {\bfseries\spaceskip=.65\fontdimen2\font\relax\xspaceskip=0pt\relax\pubtitlevalue}%
      \ifx\publinksvalue\empty
        \ifx\pubvenuevalue\empty\else\nobreak\hspace{.6em plus 1fill}\mbox{\pubrightcluster}\fi
      \else\nobreak\hspace{.6em plus 1fill}\mbox{\pubrightcluster}\fi
      \par
    \endgroup
    \begingroup
      \fontsize{9pt}{10.6pt}\selectfont\color{cvmuted}%
      \leftskip=\pubindent\rightskip=0pt plus 1fil\noindent\pubauthorsvalue\par
      \ifx\pubnotevalue\empty\else
        {\itshape\pubnotevalue}\par
      \fi
    \endgroup
    \ifx\pubsummaryvalue\empty\else
      \begingroup
        \fontsize{9.2pt}{10.8pt}\selectfont\color{cvsummaryblue}%
        \leftskip=\pubindent\rightskip=0pt plus 1fil\noindent\pubsummaryvalue\par
      \endgroup
    \fi
  \end{minipage}\par
}
